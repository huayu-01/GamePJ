using System;
using System.Collections.Generic;
using System.Linq;
using ActRoguelike.Config;
using ActRoguelike.Core;
using Godot;

namespace ActRoguelike.UI;

/// <summary>书页式内容图鉴：分类、搜索、悬停选择和独立详情；只在打开或选择时刷新。</summary>
public partial class CodexScreen : OverlayScreen
{
	public const Key DefaultToggleKey = Key.J;
	protected override Control Root => _root;
	private Control _root = null!;
	private GridContainer _grid = null!;
	private VBoxContainer _categories = null!;
	private LineEdit _search = null!;
	private Label _name = null!, _description = null!, _count = null!, _chapter = null!, _empty = null!;
	private RichTextLabel _information = null!;
	private TextureRect _icon = null!;
	private Label _title = null!;
	private CodexModelPreview _preview = null!;
	private ScrollContainer _detailScroll = null!, _listScroll = null!;
	private readonly Dictionary<string, Button> _tiles = new(), _tabs = new();
	private IReadOnlyList<CodexEntry> _entries = Array.Empty<CodexEntry>();
	private List<CodexEntry> _visible = new();
	public string ActiveCategory { get; private set; } = "all";
	public string SelectedKey { get; private set; } = "";
	public int EntryCount => _entries.Count;
	public int VisibleEntryCount => _visible.Count;
	public string SelectedName => _name.Text;
	public string SelectedDescription => _description.Text;
	public bool ModelVisible => _preview.HasModel && _preview.Visible;
	public string ModelPath => _preview.CurrentModelPath;
	public LineEdit SearchBox => _search;
	public IReadOnlyList<CodexEntry> Entries => _entries;
	private static Color Ink => GlassUi.Ink;
	private static Color Gold => GlassUi.Secondary;
	private static string T(string key, string fallback) => CodexCatalog.Text(key, fallback);
	public override void _Ready()
	{
		Layer = 18;
		BuildLayout(); InitializeOverlay();
		GlassUi.StyleKeyHints(_root);
		SetProcess(false);
	}
	private static Label Label(string text) => new() { Text = text, MouseFilter = Control.MouseFilterEnum.Ignore, AutowrapMode = TextServer.AutowrapMode.WordSmart };
	private static void FontSize(Control control, int size) => control.AddThemeFontSizeOverride("font_size", size);
	private static StyleBoxFlat Surface(bool selected = false) => GlassUi.Tile(selected);
	private static Button Button(string text)
	{
		var button = new Button { Text = text, CustomMinimumSize = new Vector2(0, 36) };
		button.AddThemeStyleboxOverride("normal", Surface()); button.AddThemeStyleboxOverride("hover", Surface(true));
		button.AddThemeStyleboxOverride("pressed", Surface(true)); button.AddThemeStyleboxOverride("focus", Surface(true));
		return button;
	}
	private void BuildLayout()
	{
		_root = new Control { Name = "Root", Theme = RuntimeResources.Load<Theme>("res://Assets/UI/Themes/ui_theme.tres") }; AddChild(_root); _root.SetAnchorsAndOffsetsPreset(Control.LayoutPreset.FullRect);
		GlassUi.Prepare(_root);
		var shade = new ColorRect { Color = new Color(.018f,.025f,.035f,.12f), MouseFilter = Control.MouseFilterEnum.Stop }; _root.AddChild(shade); shade.SetAnchorsAndOffsetsPreset(Control.LayoutPreset.FullRect);
		var margin = new MarginContainer { Name = "Book" }; _root.AddChild(margin); margin.SetAnchorsAndOffsetsPreset(Control.LayoutPreset.FullRect);
		margin.AddThemeConstantOverride("margin_left", 32); margin.AddThemeConstantOverride("margin_right", 32); margin.AddThemeConstantOverride("margin_top", 24); margin.AddThemeConstantOverride("margin_bottom", 82);
		var book = new VBoxContainer(); book.AddThemeConstantOverride("separation", 14); margin.AddChild(book);
		var header = new HBoxContainer(); book.AddChild(header);
		_title = Label(T("title", "旅途图鉴")); FontSize(_title, 30); _title.Modulate = Colors.White; _title.SizeFlagsHorizontal = Control.SizeFlags.ExpandFill; header.AddChild(_title);
		_title.AddThemeColorOverride("font_color",Colors.White);
		var close = Button("×"); close.CustomMinimumSize = new Vector2(48, 40); close.Pressed += Close; header.AddChild(close);
		var body = new HBoxContainer { SizeFlagsVertical = Control.SizeFlags.ExpandFill }; body.AddThemeConstantOverride("separation", 20); book.AddChild(body);
		var nav = new ScrollContainer { CustomMinimumSize = new Vector2(166, 0), HorizontalScrollMode = ScrollContainer.ScrollMode.Disabled }; body.AddChild(nav);
		_categories = new VBoxContainer { SizeFlagsHorizontal = Control.SizeFlags.ExpandFill }; _categories.AddThemeConstantOverride("separation", 4); nav.AddChild(_categories);
		GlassUi.Background(nav);
		var middle = new VBoxContainer { SizeFlagsHorizontal = Control.SizeFlags.ExpandFill, SizeFlagsStretchRatio = 1 }; middle.CustomMinimumSize = new Vector2(280, 0); body.AddChild(middle);
		_search = new LineEdit { Name = "Search", PlaceholderText = T("search", "搜索名称、效果或前置…"), ClearButtonEnabled = true, CustomMinimumSize = new Vector2(0, 44) }; middle.AddChild(_search);
		_search.TextChanged += _ => RefreshGrid();
		_search.FocusEntered += UpdateKeyHints; _search.FocusExited += UpdateKeyHints;
		GlassUi.Background(middle);
		_count = Label(""); FontSize(_count, 14); _count.Modulate = Gold; middle.AddChild(_count);
		_listScroll = new ScrollContainer { SizeFlagsVertical = Control.SizeFlags.ExpandFill, HorizontalScrollMode = ScrollContainer.ScrollMode.Disabled }; middle.AddChild(_listScroll);
		_grid = new GridContainer { Columns = 3, SizeFlagsHorizontal = Control.SizeFlags.ExpandFill }; _grid.AddThemeConstantOverride("h_separation", 8); _grid.AddThemeConstantOverride("v_separation", 8); _listScroll.AddChild(_grid);
		_listScroll.Resized += FitColumns;
		_empty = Label(T("empty", "没有匹配的条目，试试其他关键词。")); middle.AddChild(_empty);
		var page = new PanelContainer { SizeFlagsHorizontal = Control.SizeFlags.ExpandFill, SizeFlagsStretchRatio = 1.05f, CustomMinimumSize = new Vector2(340, 0) }; page.AddThemeStyleboxOverride("panel", Surface()); body.AddChild(page);
		GlassUi.EmptyPanel(page); GlassUi.Background(page);
		_detailScroll = new ScrollContainer { HorizontalScrollMode = ScrollContainer.ScrollMode.Disabled }; page.AddChild(_detailScroll);
		var detail = new VBoxContainer { SizeFlagsHorizontal = Control.SizeFlags.ExpandFill }; detail.AddThemeConstantOverride("separation", 14); _detailScroll.AddChild(detail);
		_chapter = Label(""); FontSize(_chapter, 14); _chapter.Modulate = Gold; detail.AddChild(_chapter);
		_name = Label(""); FontSize(_name, 28); detail.AddChild(_name);
		var hero = new Control { CustomMinimumSize = new Vector2(0, 210) }; detail.AddChild(hero);
		_icon = new TextureRect { ExpandMode = TextureRect.ExpandModeEnum.IgnoreSize, StretchMode = TextureRect.StretchModeEnum.KeepAspectCentered, MouseFilter = Control.MouseFilterEnum.Ignore }; hero.AddChild(_icon); _icon.SetAnchorsAndOffsetsPreset(Control.LayoutPreset.FullRect); _icon.OffsetLeft = _icon.OffsetTop = 20; _icon.OffsetRight = _icon.OffsetBottom = -20;
		_preview = new CodexModelPreview(); hero.AddChild(_preview); _preview.SetAnchorsAndOffsetsPreset(Control.LayoutPreset.FullRect);
		_description = Label(""); FontSize(_description, 19); detail.AddChild(_description);
		detail.AddChild(new HSeparator());
		_information = new RichTextLabel { BbcodeEnabled=true,FitContent=true,ScrollActive=false,MouseFilter=Control.MouseFilterEnum.Ignore,SizeFlagsHorizontal=Control.SizeFlags.ExpandFill }; _information.AddThemeFontSizeOverride("normal_font_size",18); detail.AddChild(_information);
	}
	private void FitColumns() => _grid.Columns = Math.Max(1, (int)((_listScroll.Size.X - 12) / 126));
	public void Open()
	{
		if (IsOpen || UiRouter.Instance?.TopScreen is { } top && top is not SettingsScreen) return;
		_entries = CodexCatalog.Build();
		_title.Text = T("title", "旅途图鉴"); _search.PlaceholderText = T("search", "搜索名称、效果或前置…");
		RefreshCategories(); RefreshGrid();
		UpdateKeyHints();
		BeginOpen(); Callable.From(FitColumns).CallDeferred();
	}
	private void UpdateKeyHints()
	{
		if (_search.HasFocus()) SetKeyHints((Key.Escape,T("close","返回／关闭")),KeyHint.Text("输入","搜索"),KeyHint.Text("Tab","切换焦点"));
		else SetKeyHints(KeyHint.Of(DefaultToggleKey,T("close","关闭")),(Key.Escape,T("close","返回／关闭")),KeyHint.Text(T("direction_keys","方向键"),T("browse","浏览条目")),KeyHint.Text("悬停／点击","详情"),KeyHint.Text("滚轮","滚动"));
	}
	public override void Close()
	{
		if (!IsOpen || InputBlocked) return;
		_preview.Clear(); BeginClose();
	}
	public override void _UnhandledKeyInput(InputEvent @event)
	{
		if (@event is not InputEventKey { Pressed: true, Echo: false } key || (key.PhysicalKeycode != DefaultToggleKey && key.Keycode != DefaultToggleKey)) return;
		if (_search.HasFocus()) return;
		if (IsOpen) { if (UiRouter.Instance?.IsTopScreen(this) != false) Close(); }
		else if (UiRouter.Instance?.AnyOpen != true) Open();
		GetViewport().SetInputAsHandled();
	}
	public void ChooseCategory(string category)
	{
		if (!CodexCatalog.Categories.Contains(category)) return;
		ActiveCategory = category; RefreshGrid();
		foreach (var pair in _tabs) pair.Value.AddThemeStyleboxOverride("normal", Surface(pair.Key == category));
	}
	public void SetSearch(string query) { _search.Text = query; RefreshGrid(); }
	private void RefreshCategories()
	{
		foreach (Node child in _categories.GetChildren()) { _categories.RemoveChild(child); child.QueueFree(); } _tabs.Clear();
		foreach (string category in CodexCatalog.Categories)
		{
			int count = category == "all" ? _entries.Count : _entries.Count(e => e.Category == category); if (count == 0) continue;
			var tab = Button(CodexCatalog.CategoryName(category) + "  " + count); tab.Name = "Category_" + category; tab.Alignment = HorizontalAlignment.Left;
			tab.Pressed += () => { if (!InputBlocked) ChooseCategory(category); }; _categories.AddChild(tab); _tabs[category] = tab;
			tab.AddThemeStyleboxOverride("normal", Surface(category == ActiveCategory));
		}
	}
	private void RefreshGrid()
	{
		foreach (Node child in _grid.GetChildren()) { _grid.RemoveChild(child); child.QueueFree(); } _tiles.Clear();
		_visible = CodexCatalog.Filter(_entries, ActiveCategory, _search.Text).ToList();
		_count.Text = string.Format(T("count", "{0} · {1} 项 / 共 {2} 项"), CodexCatalog.CategoryName(ActiveCategory), _visible.Count, _entries.Count);
		_empty.Visible = _visible.Count == 0;
		foreach (var entry in _visible)
		{
			var tile = Button(""); tile.Name = "Entry" + _tiles.Count; tile.CustomMinimumSize = new Vector2(118, 124); tile.SizeFlagsHorizontal = Control.SizeFlags.ExpandFill;
			tile.TooltipText = entry.Name; _grid.AddChild(tile); _tiles[entry.Key] = tile;
			var box = new VBoxContainer { MouseFilter = Control.MouseFilterEnum.Ignore }; tile.AddChild(box); box.SetAnchorsAndOffsetsPreset(Control.LayoutPreset.FullRect); box.OffsetLeft = box.OffsetTop = 10; box.OffsetRight = box.OffsetBottom = -10;
			box.AddChild(new TextureRect { Texture = entry.Icon, CustomMinimumSize = new Vector2(48, 56), ExpandMode = TextureRect.ExpandModeEnum.IgnoreSize, StretchMode = TextureRect.StretchModeEnum.KeepAspectCentered, MouseFilter = Control.MouseFilterEnum.Ignore, Modulate = CodexCatalog.CategoryColor(entry.Category) });
			var caption = Label(entry.Name); FontSize(caption, 16); caption.HorizontalAlignment = HorizontalAlignment.Center; caption.SizeFlagsVertical = Control.SizeFlags.ExpandFill; box.AddChild(caption);
			tile.MouseEntered += () => SelectEntry(entry.Key); tile.FocusEntered += () => { SelectEntry(entry.Key); _listScroll.EnsureControlVisible(tile); }; tile.Pressed += () => SelectEntry(entry.Key);
		}
		FitColumns(); _listScroll.ScrollVertical = 0;
		if (_visible.Count > 0) SelectEntry(_visible.Any(e => e.Key == SelectedKey) ? SelectedKey : _visible[0].Key, true);
		else { SelectedKey = ""; _name.Text = T("no_selection", "暂无条目"); _chapter.Text = _description.Text = ""; _information.Text=""; _icon.Texture = null; _preview.Clear(); }
	}
	public void SelectEntry(string key, bool force = false)
	{
		if (!force && SelectedKey == key) return;
		var entry = _visible.FirstOrDefault(e => e.Key == key); if (entry == null) return;
		SelectedKey = key; _chapter.Text = CodexCatalog.CategoryName(entry.Category); _name.Text = entry.Name;
		_name.AddThemeColorOverride("font_color",CodexCatalog.CategoryColor(entry.Category));
		_description.Text = string.IsNullOrWhiteSpace(entry.Description) ? T("no_description", "详细资料待补充。") : entry.Description;
		_information.Text = GlassUi.RulesMarkup(entry.Information); _icon.Texture = entry.Icon; _icon.Modulate=CodexCatalog.CategoryColor(entry.Category); _icon.Visible = !_preview.ShowEntry(entry);
		foreach (var pair in _tiles) pair.Value.AddThemeStyleboxOverride("normal", Surface(pair.Key == key));
		_detailScroll.ScrollVertical = 0;
	}
}
